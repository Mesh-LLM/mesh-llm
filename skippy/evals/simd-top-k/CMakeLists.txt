cmake_minimum_required(VERSION 3.20)
project(sampler_simd_experiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_STAGE_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(SIMD_BACKEND "metal" CACHE STRING "Admitted backend: metal or cuda")
option(SKIPPY_EVAL_COMMON_SAMPLER "Use the upstream-backed public Skippy ABI" ON)
option(SIMD_CONTROLLER_ONLY "Build orchestration without rebuilding the frozen workers" OFF)
if(SIMD_CONTROLLER_ONLY)
    add_executable(simd-sample-stacks sample-stacks.cpp)
    target_include_directories(simd-sample-stacks SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
    target_compile_options(simd-sample-stacks PRIVATE -Wall -Wextra -Werror)
    add_executable(simd-controller controller.cpp host-observation.cpp)
    target_include_directories(simd-controller SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
    target_compile_options(simd-controller PRIVATE -Wall -Wextra -Werror)
    add_executable(simd-audit audit.cpp)
    target_include_directories(simd-audit SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
    target_compile_options(simd-audit PRIVATE -Wall -Wextra -Werror)
    add_executable(simd-contracts contracts.cpp host-observation.cpp)
    target_include_directories(simd-contracts SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
    target_compile_options(simd-contracts PRIVATE -Wall -Wextra -Werror)
    enable_testing()
    add_test(NAME simd-orchestration-contracts COMMAND simd-contracts $<TARGET_FILE:simd-controller> $<TARGET_FILE:simd-audit>)
    set_tests_properties(simd-orchestration-contracts PROPERTIES TIMEOUT 900)
    return()
endif()
if(SIMD_BACKEND STREQUAL "metal" AND APPLE)
    enable_language(OBJC)
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
    set(GGML_CUDA OFF CACHE BOOL "" FORCE)
    set(THERMAL_SOURCE ../grammar-sampling/thermal.m)
    set(THERMAL_LIBRARIES "-framework Foundation")
elseif(SIMD_BACKEND STREQUAL "cuda" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
    set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ON CACHE BOOL "" FORCE)
    find_package(CUDAToolkit REQUIRED)
    if(NOT TARGET CUDA::nvml)
        message(FATAL_ERROR "NVML is required for the Spark thermal guard")
    endif()
    set(THERMAL_SOURCE thermal-cuda.cpp)
    set(THERMAL_LIBRARIES CUDA::nvml)
else()
    message(FATAL_ERROR "Only admitted M6/Metal and Spark/Linux/CUDA configurations are supported")
endif()
enable_testing()
add_executable(simd-thermal-probe thermal-probe.cpp ${THERMAL_SOURCE})
target_link_libraries(simd-thermal-probe PRIVATE ${THERMAL_LIBRARIES})
add_subdirectory("${LLAMA_ROOT}" native)
add_executable(grammar-sampling ../grammar-sampling/runner.cpp ../grammar-sampling/fixtures.cpp ${THERMAL_SOURCE})
target_include_directories(grammar-sampling SYSTEM PRIVATE "${LLAMA_ROOT}/src" "${LLAMA_ROOT}/common" "${LLAMA_ROOT}/vendor/nlohmann")
target_link_libraries(grammar-sampling PRIVATE llama-common llama ${THERMAL_LIBRARIES})
target_compile_options(grammar-sampling PRIVATE -Wall -Wextra -Werror)
if(SKIPPY_EVAL_COMMON_SAMPLER)
    add_executable(grammar-verification ../grammar-sampling/verification.cpp ../grammar-sampling/fixtures.cpp)
    target_include_directories(grammar-verification SYSTEM PRIVATE "${LLAMA_ROOT}/src" "${LLAMA_ROOT}/common" "${LLAMA_ROOT}/vendor/nlohmann")
    target_link_libraries(grammar-verification PRIVATE llama-common llama)
    target_compile_options(grammar-verification PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(grammar-sampling PRIVATE SKIPPY_EVAL_COMMON_SAMPLER=1)
    target_compile_definitions(grammar-verification PRIVATE SKIPPY_EVAL_COMMON_SAMPLER=1)
endif()
