cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Windows")
    message(FATAL_ERROR "KV validation supports Linux and macOS only: packaged Windows runtimes do not export the private graph inspection helpers")
endif()
project(indexed_real_kv_matrix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(NATIVE_SOURCE "" CACHE PATH "Matching prepared native source checkout")
set(NATIVE_LIB_DIR "" CACHE PATH "Exact final runtime library directory")
if(NOT EXISTS "${NATIVE_SOURCE}/src/llama-context.h")
    message(FATAL_ERROR "NATIVE_SOURCE must name the matching prepared source")
endif()
foreach(component llama ggml ggml-base)
    find_library(NATIVE_${component} NAMES ${component} PATHS "${NATIVE_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(indexed-real-kv-matrix indexed-real-kv-matrix.cpp)
target_compile_definitions(indexed-real-kv-matrix PRIVATE KV_NATIVE_LIB_DIR="${NATIVE_LIB_DIR}")
target_include_directories(indexed-real-kv-matrix PRIVATE "${NATIVE_SOURCE}/include"
    "${NATIVE_SOURCE}/src" "${NATIVE_SOURCE}/ggml/include" "${NATIVE_SOURCE}/ggml/src")
target_link_libraries(indexed-real-kv-matrix PRIVATE ${NATIVE_llama} ${NATIVE_ggml} ${NATIVE_ggml-base})
set_target_properties(indexed-real-kv-matrix PROPERTIES BUILD_RPATH "${NATIVE_LIB_DIR}")
