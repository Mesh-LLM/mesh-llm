cmake_minimum_required(VERSION 3.20)
project(grammar_sampling_abc LANGUAGES C CXX OBJC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_STAGE_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_ACCELERATE ON CACHE BOOL "" FORCE)
enable_testing()
add_subdirectory("${LLAMA_ROOT}" native)
add_executable(grammar-controller controller.cpp)
target_include_directories(grammar-controller SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
target_compile_options(grammar-controller PRIVATE -Wall -Wextra -Werror)
add_executable(grammar-audit audit.cpp)
target_include_directories(grammar-audit SYSTEM PRIVATE "${LLAMA_ROOT}/vendor/nlohmann")
target_compile_options(grammar-audit PRIVATE -Wall -Wextra -Werror)
add_executable(grammar-sampling runner.cpp fixtures.cpp thermal.m)
target_include_directories(grammar-sampling SYSTEM PRIVATE "${LLAMA_ROOT}/src" "${LLAMA_ROOT}/common" "${LLAMA_ROOT}/vendor/nlohmann")
target_link_libraries(grammar-sampling PRIVATE llama-common llama "-framework Foundation")
target_compile_options(grammar-sampling PRIVATE -Wall -Wextra -Werror)
add_executable(grammar-verification verification.cpp fixtures.cpp)
target_include_directories(grammar-verification SYSTEM PRIVATE "${LLAMA_ROOT}/src" "${LLAMA_ROOT}/common" "${LLAMA_ROOT}/vendor/nlohmann")
target_link_libraries(grammar-verification PRIVATE llama-common llama)
target_compile_options(grammar-verification PRIVATE -Wall -Wextra -Werror)
